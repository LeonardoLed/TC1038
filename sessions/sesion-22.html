Python · Ejemplo / punto de partida
nombre = input("Jugador: ")
puntaje = int(input("Puntaje: "))
with open("puntajes.txt", "a", encoding="utf-8") as archivo:
    archivo.write(f"{nombre},{puntaje}\n")
